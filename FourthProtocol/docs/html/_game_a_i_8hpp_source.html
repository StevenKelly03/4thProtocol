130int Game::evaluateLine(int a, int b, int c, int d, int aiPlayer) const
131{
132 int indices[4] = { a, b, c, d };
133
134 int aiCount = 0;
135 int oppCount = 0;
136
137 for (int i = 0; i < 4; ++i)
138 {
139 Piece* p = board[indices[i]];
140 if (!p) continue;
141 if (p->getOwner() == aiPlayer)
142 ++aiCount;
143 else
144 ++oppCount;
145 }
146
147 if (aiCount > 0 && oppCount > 0)
148 return 0; // blocked line
149
150 int score = 0;
151
152 if (aiCount > 0)
153 {
154 if (aiCount == 1) score = 1;
155 else if (aiCount == 2) score = 10;
156 else if (aiCount == 3) score = 50;
157 else if (aiCount == 4) score = 1000;
158 }
159 else if (oppCount > 0)
160 {
161 if (oppCount == 1) score = -1;
162 else if (oppCount == 2) score = -10;
163 else if (oppCount == 3) score = -50;
164 else if (oppCount == 4) score = -1000;
165 }
166
167 return score;
168}

170int Game::evaluateBoard(int aiPlayer) const
171{
172 int score = 0;
173 int opponent = (aiPlayer == 1 ? 2 : 1);
174
175 // --- Piece type values ---
176 auto pieceValue = [&](Piece* p)
177 {
178 if (!p) return 0;
179 switch (p->getType())
180 {
181 case PieceType::Frog: return 15;
182 case PieceType::Snake: return 10;
183 case PieceType::Donkey: return 3;
184 }
185 return 0;
186 };
187
188 // --- Center control values ---
189 const int centerSquares[5] = { 6, 7, 8, 11, 12 };
190
191 // --- Mobility: number of available moves ---
192 std::vector<Move> aiMoves, oppMoves;
193 generateAllMoves(aiPlayer, aiMoves);
194 generateAllMoves(opponent, oppMoves);
195
196 score += aiMoves.size() * 2;
197 score -= oppMoves.size() * 2;
198
199 // --- Piece values & center control ---
200 for (int i = 0; i < 25; i++)
201 {
202 Piece* p = board[i];
203 if (!p) continue;
204
205 int base = pieceValue(p);
206
207 if (p->getOwner() == aiPlayer)
208 {
209 score += base;
210
211 // center bonus
212 for (int c : centerSquares)
213 if (i == c)
214 score += 3;
215 }
216 else
217 {
218 score -= base;
219
220 for (int c : centerSquares)
221 if (i == c)
222 score -= 3;
223 }
224 }
225
226 // ---- Advanced line scoring (4-in-a-row windows) ----
227 auto scoreLine = [&](int a, int b, int c, int d)
228 {
229 return evaluateLine(a, b, c, d, aiPlayer);
230 };
231
232 // horizontal
233 for (int r = 0; r < 5; r++)
234 for (int c = 0; c <= 1; c++)
235 score += scoreLine(r * 5 + c, r * 5 + c + 1, r * 5 + c + 2, r * 5 + c + 3);
236
237 // vertical
238 for (int c = 0; c < 5; c++)
239 for (int r = 0; r <= 1; r++)
240 score += scoreLine(r * 5 + c, (r + 1) * 5 + c, (r + 2) * 5 + c, (r + 3) * 5 + c);
241
242 // diag down-right
243 for (int r = 0; r <= 1; r++)
244 for (int c = 0; c <= 1; c++)
245 score += scoreLine(r * 5 + c, (r + 1) * 5 + c + 1, (r + 2) * 5 + c + 2, (r + 3) * 5 + c + 3);
246
247 // diag up-right
248 for (int r = 3; r < 5; r++)
249 for (int c = 0; c <= 1; c++)
250 score += scoreLine(r * 5 + c, (r - 1) * 5 + c + 1, (r - 2) * 5 + c + 2, (r - 3) * 5 + c + 3);
251
252 return score;
253}

260int Game::minimax(int depth, int maxDepth, bool maximizingPlayer, int aiPlayer,
261 int alpha, int beta)
262{
263 int winner = checkWinner();
264 if (winner != 0)
265 {
266 if (winner == aiPlayer)
267 return 100000 - depth;
268 else
269 return -100000 + depth;
270 }
271
272 if (depth >= maxDepth || isBoardFull())
273 return evaluateBoard(aiPlayer);
274
275 int currentPlayer = maximizingPlayer ? aiPlayer : (aiPlayer == 1 ? 2 : 1);
276
277 std::vector<Move> moves;
278 generateAllMoves(currentPlayer, moves);
279
280 if (moves.empty())
281 return evaluateBoard(aiPlayer);
282
283 if (maximizingPlayer)
284 {
285 int best = -1000000;
286
287 for (auto& m : moves)
288 {
289 applyMoveToBoardOnly(m.from, m.to);
290 int val = minimax(depth + 1, maxDepth, false, aiPlayer, alpha, beta);
291 undoMoveOnBoardOnly(m.from, m.to);
292
293 best = std::max(best, val);
294 alpha = std::max(alpha, best);
295
296 // PRUNE HERE
297 if (beta <= alpha)
298 break;
299 }
300
301 return best;
302 }
303 else
304 {
305 int best = 1000000;
306
307 for (auto& m : moves)
308 {
309 applyMoveToBoardOnly(m.from, m.to);
310 int val = minimax(depth + 1, maxDepth, true, aiPlayer, alpha, beta);
311 undoMoveOnBoardOnly(m.from, m.to);
312
313 best = std::min(best, val);
314 beta = std::min(beta, best);
315
316 // PRUNE HERE
317 if (beta <= alpha)
318 break;
319 }
320
321 return best;
322 }
323}
